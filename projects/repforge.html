app/src/main/java/com/repforge/app/domain/metrics/ProgressMetrics.kt
fun progressSeries(entries: List<HistoryEntry>, zone: ZoneId): List<ProgressPoint> {
    val work = entries.working()
    val kind = work.firstOrNull()?.kind ?: KIND_REPS
    val valued = when (kind) {
        KIND_TIMED -> work.mapNotNull { e -> e.durationSec?.takeIf { it > 0 }?.let { e to it.toDouble() } }
        KIND_CARDIO -> work.mapNotNull { e -> e.distanceKm?.takeIf { it > 0 }?.let { e to it } }
        else -> work.filter { it.reps > 0 && it.loaded }.map { it to estimatedOneRepMax(it.weightKg, it.reps) }
    }
    return valued.groupBy({ localDate(it.first.recordedAt, zone) }, { it.second })
        .map { (date, day) -> ProgressPoint(date, day.max()) }
        .sortedBy { it.date }
}

app/src/main/java/com/repforge/app/domain/metrics/WorkoutMetrics.kt
fun estimatedOneRepMax(weightKg: Double, reps: Int): Double {
    require(weightKg >= 0) { "Carga não pode ser negativa." }
    require(reps > 0) { "Repetições devem ser maiores que zero." }
    return weightKg * (1.0 + reps / 30.0)
}

A fórmula estima o 1RM a partir de carga e repetições. É uma estimativa, não um teste de força máxima.The formula estimates a 1RM from load and reps. It is an estimate, not a maximal strength test.
